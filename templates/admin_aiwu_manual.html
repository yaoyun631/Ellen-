愛屋手動更新工具（JS 抓網址 + 上傳 TXT）

  1. 先在新分頁打開愛屋登入頁：
    https://es.houseol.com.tw/login.aspx
  2. 登入帳號（H229 / sp290 / 0000），進到出售物件列表頁。
  3. 按 F12 開啟開發者工具 → 切到「Console」。
  4. 貼上下面這段 JS，按 Enter 執行，等它自動點完「查看更多」，會自動下載 愛屋YYYYMMDD.txt：
function autoClickAndExportUniqueUrls(interval = 3000) {
  const timer = setInterval(() => {
    window.scrollTo({
      top: document.body.scrollHeight,
      behavior: 'smooth'
    });

    const loadMoreBtn = document.querySelector('a.load_more');

    if (loadMoreBtn) {
      console.log("找到『查看更多』按鈕，準備點擊...");
      loadMoreBtn.click();
    } else {
      console.log("沒有找到『查看更多』按鈕，停止自動點擊。");
      clearInterval(timer);

      setTimeout(() => {
        const catalogLinks = Array.from(document.querySelectorAll('a'))
          .filter(a => a.textContent.trim() === '型錄' && a.href.includes('Ecatalog.aspx'));

        const uniqueLinks = [...new Set(catalogLinks.map(link =>
          link.href.startsWith('/') ? location.origin + link.href : link.href
        ))];

        const txtContent = uniqueLinks.join('\n');

        const now = new Date();
        const yyyy = now.getFullYear();
        const mm = String(now.getMonth() + 1).padStart(2, '0');
        const dd = String(now.getDate()).padStart(2, '0');
        const filename = `愛屋${yyyy}${mm}${dd}.txt`;

        const blob = new Blob([txtContent], { type: 'text/plain;charset=utf-8;' });

        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = filename;
        a.style.display = 'none';
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);

        console.log(`✅ 已匯出 ${uniqueLinks.length} 筆「不重複」的型錄網址至 ${filename}`);
      }, 2000);
    }
  }, interval);
}

autoClickAndExportUniqueUrls();
